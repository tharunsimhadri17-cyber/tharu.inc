//================ Countdown ==================

const countdown=document.getElementById("countdown");

const target=new Date("August 2, 2026 00:00:00").getTime();

function timer(){

const now=new Date().getTime();

let diff=target-now;

if(diff<=0){

countdown.innerHTML="🎉 IT'S BIRTHDAY TIME 🎉";

return;

}

let d=Math.floor(diff/86400000);
let h=Math.floor((diff%86400000)/3600000);
let m=Math.floor((diff%3600000)/60000);
let s=Math.floor((diff%60000)/1000);

countdown.innerHTML=
`${d}d ${h}h ${m}m ${s}s`;

}

setInterval(timer,1000);
timer();

//================ Confetti ==================

const canvas=document.getElementById("confetti");
const ctx=canvas.getContext("2d");

canvas.width=innerWidth;
canvas.height=innerHeight;

let pieces=[];

for(let i=0;i<200;i++){

pieces.push({
x:Math.random()*canvas.width,
y:Math.random()*canvas.height,
r:Math.random()*6+2,
d:Math.random()*5+1,
c:`hsl(${Math.random()*360},100%,60%)`
});

}

function draw(){

ctx.clearRect(0,0,canvas.width,canvas.height);

pieces.forEach(p=>{

ctx.beginPath();

ctx.fillStyle=p.c;

ctx.arc(p.x,p.y,p.r,0,Math.PI*2);

ctx.fill();

p.y+=p.d;

if(p.y>canvas.height)p.y=0;

});

requestAnimationFrame(draw);

}

draw();

//================ Puzzle ==================

let board=[1,2,3,4,5,6,7,8,""];

board.sort(()=>Math.random()-0.5);

function render(){

const p=document.getElementById("puzzle");

p.innerHTML="";

board.forEach((v,i)=>{

let d=document.createElement("div");

d.className="tile";

if(v==="") d.classList.add("empty");

d.innerHTML=v;

d.onclick=()=>move(i);

p.appendChild(d);

});

}

function move(i){

let empty=board.indexOf("");

let allowed=[i-1,i+1,i-3,i+3];

if(allowed.includes(empty)){

[board[i],board[empty]]=[board[empty],board[i]];

render();

if(board.join()=="1,2,3,4,5,6,7,8,")

setTimeout(()=>alert("🎉 Puzzle Solved!"),100);

}

}
